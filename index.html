const express = require('express'); const sqlite3 = require('sqlite3').verbose(); const path = require('path'); const app = express(); const PORT = process.env.PORT || 3000; const ADMIN_PASSWORD = 'mtfootball-admin'; app.use(express.json()); app.use(express.static('public')); // اتصال به دیتابیس SQLite const db = new sqlite3.Database('./teams.db', (err) => { if (err) console.error(err); else console.log('Connected to SQLite database.'); }); // ساخت جدول تیم‌ها با بک‌تیک db.run(CREATE TABLE IF NOT EXISTS teams ( id INTEGER PRIMARY KEY AUTOINCREMENT, name TEXT UNIQUE, status TEXT DEFAULT 'free', coach TEXT DEFAULT '' )); // اضافه کردن تیم‌های لیگ ایران const iranLeagueTeams = [ 'پرسپولیس','استقلال','سپاهان','تراکتور','فولاد','ذوب‌آهن','گل‌گهر', 'آلومینیوم','ملوان','نساجی','پیکان','هوادار', 'استقلال خوزستان','صنعت نفت','مس رفسنجان','شمس‌آذر' ]; iranLeagueTeams.forEach(team => { db.run(INSERT OR IGNORE INTO teams (name) VALUES (?), [team]); }); // گرفتن تیم‌ها app.get('/teams', (req, res) => { db.all("SELECT * FROM teams", [], (err, rows) => { if (err) return res.status(500).json({success:false, message: err.message}); res.json(rows); }); }); // انتخاب تیم توسط مربی app.post('/select', (req, res) => { const { team, coach } = req.body; db.run( UPDATE teams SET status='taken', coach=? WHERE name=? AND status='free', [coach, team], function(err){ if(err) return res.status(500).json({success:false, message: err.message}); if(this.changes === 0) return res.json({success:false, message:'Team already taken'}); res.json({success:true}); } ); }); // عملیات ادمین app.post('/admin', (req, res) => { const { team, action, password, name } = req.body; if(password !== ADMIN_PASSWORD) return res.json({success:false, message:'Wrong password'}); let sql = '', params = []; if(action === 'reserve'){ sql = UPDATE teams SET status='reserved', coach=? WHERE name=?; params = [name||'', team]; } else if(action === 'lock_admin'){ sql = UPDATE teams SET status='locked_admin' WHERE name=?; params = [team]; } else if(action === 'free'){ sql = UPDATE teams SET status='free', coach='' WHERE name=?; params = [team]; } else return res.json({success:false,message:'Invalid action'}); db.run(sql, params, function(err){ if(err) return res.status(500).json({success:false,message:err.message}); res.json({success:true}); }); }); // شروع سرور app.listen(PORT, () => console.log(Server running at http://localhost:${PORT}));
